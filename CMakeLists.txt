cmake_minimum_required(VERSION 3.21)
project(Starlet LANGUAGES C CXX)

option(STARLET_GLES "Link the GX renderer against the target's OpenGL ES 3.1 library" OFF)
option(STARLET_TESTS "Build Starlet's tests" ${PROJECT_IS_TOP_LEVEL})

if(STARLET_TESTS)
    enable_testing()
endif()

add_subdirectory(src/gx)
add_subdirectory(src/disc)
add_subdirectory(src/card)
add_subdirectory(src/host)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(src/os)
    foreach(sdk_module gx_fifo dvd pad ar ai ax)
        if(EXISTS "${PROJECT_SOURCE_DIR}/src/${sdk_module}/CMakeLists.txt")
            add_subdirectory(src/${sdk_module})
        endif()
    endforeach()
endif()
if(STARLET_TESTS)
    add_subdirectory(tests/gx)
    add_subdirectory(tests/disc)
    add_subdirectory(tests/card)
    add_subdirectory(tests/ai)
    add_subdirectory(tests/ax)
    if(TARGET starlet_os)
        add_subdirectory(tests/os)
    endif()
    if(TARGET starlet_gx_fifo)
        add_subdirectory(tests/gx_fifo)
    endif()
    if(TARGET starlet_dvd)
        add_subdirectory(tests/dvd)
    endif()
    if(TARGET starlet_host)
        add_subdirectory(tests/host)
    endif()
endif()
